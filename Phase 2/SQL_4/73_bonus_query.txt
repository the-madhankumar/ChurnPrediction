mysql> SELECT
    -> d.contract_type,
    -> AVG(f.monthly_charges) AS avg_monthly_charges,
    -> AVG(f.tenure) AS avg_tenure,
    -> AVG(f.churn) AS churn_rate
    -> FROM fact_customer_account AS f
    -> JOIN dim_contract AS d
    -> ON f.contract_key = d.contract_key
    -> GROUP BY d.contract_type
    -> ORDER BY churn_rate DESC;
