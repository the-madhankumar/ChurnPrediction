#129. Add a 404 handler for all routes using @app.exception_handler(404) 
@app.exception_handler(404)
async def not_found_handler(
    request: Request,
    exc
):
    error = ErrorResponse(
        detail="Resource not found",
        status_code=404
    )

    return JSONResponse(
        status_code=404,
        content=error.model_dump()
    )
